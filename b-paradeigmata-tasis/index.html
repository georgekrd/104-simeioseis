Τα παραδείγματα εφαρμόζουν αριθμητικά τις σχέσεις του κεφαλαίου. Το πρώτο εφαρμόζει τα τρία επίπεδα της ιεραρχικής ρύθμισης τάσης σε μια ζώνη πέντε ζυγών, από την απόκριση των ρυθμιστών τάσης των μονάδων έως τον προγραμματισμό της επόμενης ημέρας. Το δεύτερο εξετάζει την ευστάθεια τάσης στην ίδια ζώνη με καμπύλες PPP–VVV και QQQ–VVV, ανάλυση ιδιομορφών και προσομοίωση της αποκατάστασης του φορτίου από το ΣΑΤΥΦ. Το τρίτο υπολογίζει τη στήριξη της τάσης από μετατροπείς κατά τα σφάλματα, για διαφορετικούς λόγους βραχυκύκλωσης.

Διαδραστική εκτέλεση

Όπως και στα παραδείγματα του Κεφαλαίου 1, στην ηλεκτρονική έκδοση των σημειώσεων τα κελιά εκτελούνται στον φυλλομετρητή, ενώ στο PDF εμφανίζονται τα αποτελέσματα με τις προεπιλεγμένες τιμές. Ο πυρήνας Python ενεργοποιείται με το κουμπί έναρξης περιβάλλοντος υπολογισμού και τα κελιά κάθε ενότητας εκτελούνται με τη σειρά, επειδή χρησιμοποιούν συναρτήσεις και μεταβλητές που ορίζονται στα προηγούμενα κελιά. Η ενότητα της ευστάθειας τάσης χρησιμοποιεί επιπλέον τα δεδομένα και τις συναρτήσεις της πρώτης ενότητας, της οποίας τα κελιά εκτελούνται πρώτα.

Οι παράμετροι μεταβάλλονται με τους ολισθητές των ενοτήτων «Παραμετρική διερεύνηση με ολισθητές» και «Μακροπρόθεσμη αστάθεια από το ΣΑΤΥΦ» ή με το κουμπί ανοίγματος σε Jupyter, το οποίο φορτώνει τη σελίδα ως σημειωματάριο με δυνατότητα πλήρους επεξεργασίας του κώδικα.

Ιεραρχική ρύθμιση τάσης σε ζώνη πέντε ζυγών¶

Το παράδειγμα εφαρμόζει τα τρία επίπεδα της Ενότητας «Ιεραρχική ρύθμιση τάσης» και τα στάδια της Ενότητας «Ρύθμιση τάσης από τον προγραμματισμό στον πραγματικό χρόνο» σε μια ζώνη ρύθμισης τάσης. Η ζώνη περιλαμβάνει τους κλάδους του δικτύου τριών ζυγών των παραδειγμάτων του Κεφαλαίου 1, δύο μονάδες ηλεκτροπαραγωγής και μια συστοιχία πυκνωτών.

Η ζώνη και η ροή ισχύος¶

Ο ζυγός 1 συνδέει τη ζώνη με το υπόλοιπο σύστημα, το οποίο παριστάνεται ως πηγή σταθερής τάσης V1V_1V1​ = 1,00 p.u., και είναι ο ζυγός αναφοράς. Οι μονάδες G4 και G5 συνδέονται στους ζυγούς 2 και 3 μέσω μετασχηματιστών ανύψωσης και οι ρυθμιστές τάσης (AVR) τους διατηρούν την τάση των ακροδεκτών τους στις τιμές αναφοράς. Τα φορτία των ζυγών 2 και 3 είναι 100 MW και 35 Mvar και 120 MW και 45 Mvar αντίστοιχα. Ο ζυγός 3, ο πιο απομακρυσμένος από το υπόλοιπο σύστημα, είναι ο ζυγός-πιλότος της ζώνης και διαθέτει συστοιχία τεσσάρων βημάτων πυκνωτών των 10 Mvar, από τα οποία δύο είναι αρχικά σε λειτουργία (Σχήμα 17, Πίνακας 6).

Η ροή ισχύος επιλύεται με τη μέθοδο Newton–Raphson σε πολικές συντεταγμένες. Οι άγνωστοι είναι οι γωνίες όλων των ζυγών εκτός του ζυγού αναφοράς και τα μέτρα των τάσεων των ζυγών φορτίου, ενώ οι εξισώσεις είναι τα ισοζύγια ενεργής ισχύος των ίδιων ζυγών και τα ισοζύγια άεργης ισχύος των ζυγών φορτίου. Ο ιακωβιανός πίνακας έχει τη μορφή της (16) και υπολογίζεται σε μορφή πινάκων από τις παραγώγους των μιγαδικών εγχύσεων S~i=V~i(∑jYijV~j)∗\tilde{S}_i = \tilde{V}_i \left(\sum_j Y_{ij}\tilde{V}_j\right)^{*}S~i​=V~i​(∑j​Yij​V~j​)∗, οι οποίες συμπίπτουν με τις παραγώγους των εγχύσεων της εκτίμησης κατάστασης του Κεφαλαίου 1.

KLADOI = [

# --- φορτία (p.u.) ---

MONADES = {

def build_ybus(kladoi, n_pyk=0, ektos=()):
    """Πίνακας αγωγιμοτήτων ζυγών με n_pyk βήματα πυκνωτών στον ζυγό 3.

    ektos: κλάδοι εκτός λειτουργίας, ως ζεύγη (i, j) με αρίθμηση από το 1.
    """
    Y = np.zeros((NB, NB), dtype=complex)
    for i, j, r, x, b in kladoi:
        if (i, j) in ektos:
            continue
        i, j = i - 1, j - 1
        y = 1 / complex(r, x)
        Y[i, i] += y + 1j * b / 2
        Y[j, j] += y + 1j * b / 2
        Y[i, j] -= y
        Y[j, i] -= y
    Y[2, 2] += 1j * n_pyk * B_VIMA
    return Y


def egxyseis(Y, V, th):
    """Μιγαδικές εγχύσεις ισχύος S = V (Y V)* όλων των ζυγών."""
    Vc = V * np.exp(1j * th)
    return Vc * np.conj(Y @ Vc)


def paragogoi(Y, V, th):
    """Παράγωγοι των εγχύσεων ως προς τις γωνίες και τα μέτρα των τάσεων."""
    Vc = V * np.exp(1j * th)
    I = Y @ Vc
    dS_dth = 1j * np.diag(Vc) @ np.conj(np.diag(I) - Y @ np.diag(Vc))
    dS_dV = (np.diag(Vc) @ np.conj(Y @ np.diag(Vc / V))
             + np.diag(np.conj(I) * Vc / V))
    return dS_dth, dS_dV


def roi_isxyos(Y, V_sp, P_sp, Q_sp, pv, tol=1e-10, typose=False):
    """Ροή ισχύος Newton–Raphson σε πολικές συντεταγμένες.

    Ο ζυγός 1 (δείκτης 0) είναι ο ζυγός αναφοράς. Οι ζυγοί pv έχουν
    ελεγχόμενη τάση V_sp και οι υπόλοιποι είναι ζυγοί φορτίου.
    """
    pq = [i for i in range(1, NB) if i not in pv]
    an = list(range(1, NB))
    V, th = np.array(V_sp, float), np.zeros(NB)
    for it in range(20):
        S = egxyseis(Y, V, th)
        F = np.r_[P_sp[an] - S.real[an], Q_sp[pq] - S.imag[pq]]
        if typose:
            print(f"επανάληψη {it}: max |ΔP|, |ΔQ| = {np.abs(F).max():.2e}")
        if np.abs(F).max() < tol:
            return V, th, S
        dSt, dSv = paragogoi(Y, V, th)
        J = np.block([[dSt.real[np.ix_(an, an)], dSv.real[np.ix_(an, pq)]],
                      [dSt.imag[np.ix_(pq, an)], dSv.imag[np.ix_(pq, pq)]]])
        dx = np.linalg.solve(J, F)
        th[an] += dx[:len(an)]
        V[pq] += dx[len(an):]
    raise RuntimeError("η ροή ισχύος δεν συγκλίνει")


def katastasi(lam=1.0, n_pyk=N_PYK, V_ref=None, dP3=0.0, dQ3=0.0,
              ektos=(), xoris=(), typose=False):
    """Μόνιμη κατάσταση με τις μονάδες σε ρύθμιση τάσης (πρωτεύουσα).

    lam: συντελεστής φόρτισης, dP3 + jdQ3: πρόσθετο φορτίο στον ζυγό 3,
    V_ref: τιμές αναφοράς των AVR (διάνυσμα μήκους NB) ή None,
    ektos: κλάδοι εκτός λειτουργίας, xoris: μονάδες εκτός λειτουργίας.
    """
    PL, QL = lam * P_L, lam * Q_L
    PL[2] += dP3
    QL[2] += dQ3
    P_sp, Q_sp, V_sp = -PL, -QL, np.ones(NB)
    V_sp[0] = V1
    pv = [g for g in GEN if g not in xoris]
    for g in pv:
        P_sp[g] += MONADES[g]["P"]
        V_sp[g] = MONADES[g]["V"] if V_ref is None else V_ref[g]
    Y = build_ybus(KLADOI, n_pyk, ektos)
    V, th, S = roi_isxyos(Y, V_sp, P_sp, Q_sp, pv=pv, typose=typose)
    return V, th, S, S.imag + QL          # το τελευταίο: άεργη παραγωγή

Η μέθοδος συγκλίνει σε τέσσερις επαναλήψεις. Οι τάσεις των ζυγών φορτίου είναι λίγο κάτω από 1 p.u., και το μεγαλύτερο μέρος της άεργης ισχύος των φορτίων καλύπτεται από τις δύο μονάδες και τους πυκνωτές της ζώνης, όπως επιβάλλει ο τοπικός χαρακτήρας της ρύθμισης τάσης: η ζώνη ανταλλάσσει με το υπόλοιπο σύστημα ελάχιστη άεργη ισχύ.

Για καλύτερη εποπτεία, τα αποτελέσματα σχεδιάζονται και πάνω στο σχήμα της ζώνης με τη συνάρτηση diagramma του κελιού που ακολουθεί. Ο κώδικας του κελιού αφορά μόνο τη σχεδίαση και εμφανίζεται στην ηλεκτρονική έκδοση. Στους ζυγούς σημειώνονται τα μέτρα των τάσεων, με κόκκινο φόντο όταν βρίσκονται εκτός των ορίων 0,95–1,05 p.u. Στο μέσον κάθε κλάδου το μαύρο βέλος δείχνει τη ροή ενεργής ισχύος, ενώ στα δύο άκρα του τα κόκκινα βέλη δείχνουν την άεργη ισχύ που ρέει από τον ζυγό προς τον κλάδο ή αντίστροφα. Όταν και τα δύο κόκκινα βέλη δείχνουν προς τον κλάδο, όπως στον κλάδο 1–3 της βασικής κατάστασης, ο κλάδος απορροφά άεργη ισχύ και από τα δύο άκρα του. Στις μονάδες σημειώνονται η ενεργή και η άεργη ισχύς τους και ο λόγος Q/Qmax⁡Q/Q^{\max}Q/Qmax, ενώ η ράβδος δείχνει τη θέση της άεργης ισχύος μεταξύ των ορίων Qmin⁡Q^{\min}Qmin και Qmax⁡Q^{\max}Qmax. Όταν συγκρίνονται δύο καταστάσεις, οι μεταβολές σημειώνονται σε παρένθεση.

def megethi(V, th, n_pyk=N_PYK, lam=1.0, dP3=0.0, dQ3=0.0, ektos=(),
            xoris=()):
    """Μεγέθη μιας κατάστασης για το σχήμα (MW, Mvar, p.u.)."""
    Vc = V * np.exp(1j * th)
    S = egxyseis(build_ybus(KLADOI, n_pyk, ektos), V, th)
    PL, QL = lam * P_L, lam * Q_L
    PL[2] += dP3
    QL[2] += dQ3
    roes = {}                       # ισχύς από κάθε άκρο προς τον κλάδο
    for i, j, r, x, b in KLADOI[:3]:
        if (i, j) not in ektos:
            y, a, c = 1 / complex(r, x), i - 1, j - 1
            roes[(i, j)] = [100 * Vc[k] * np.conj((Vc[k] - Vc[n]) * y
                                                   + 0.5j * b * Vc[k])
                            for k, n in ((a, c), (c, a))]
    return dict(V=V, S=100 * S, PL=100 * PL, QL=100 * QL, roes=roes,
                Qc=100 * n_pyk * B_VIMA * V[2]**2, n_pyk=n_pyk,
                ektos=ektos, xoris=xoris)

Δευτερεύουσα ρύθμιση με ζυγό-πιλότο¶

Η δευτερεύουσα ρύθμιση εφαρμόζεται με τη (10). Ο ρυθμιστής της ζώνης υπολογίζει κάθε δευτερόλεπτο το επίπεδο άεργης ισχύος qqq από το σφάλμα της τάσης του ζυγού-πιλότου, και ένας τοπικός βρόχος σε κάθε μονάδα μεταβάλλει την τιμή αναφοράς του AVR της κατά KL (q Qimax⁡−Qi) ΔtK_L\,(q\,Q_i^{\max} - Q_i)\,\Delta tKL​(qQimax​−Qi​)Δt, ώστε η άεργη ισχύς της να ακολουθεί την Qiref=q Qimax⁡Q_i^{\mathrm{ref}} = q\,Q_i^{\max}Qiref​=qQimax​. Επειδή η πρωτεύουσα ρύθμιση είναι πολύ γρηγορότερη, σε κάθε βήμα η κατάσταση υπολογίζεται με ροή ισχύος για τις τρέχουσες τιμές αναφοράς των AVR. Η προσομοίωση αυτή, με διαδοχικές καταστάσεις ισορροπίας, ονομάζεται οιονεί μόνιμη (quasi steady-state) και είναι κατάλληλη για φαινόμενα που εξελίσσονται σε δεκάδες δευτερόλεπτα ή λεπτά.

Η αρχική κατάσταση είναι η μόνιμη κατάσταση της δευτερεύουσας ρύθμισης για τάση αναφοράς του ζυγού-πιλότου 0,99 p.u. Υπολογίζεται απευθείας με μια παραλλαγή της ροής ισχύος, στην οποία οι ζυγοί των μονάδων έχουν άεργη ισχύ q Qimax⁡q\,Q_i^{\max}qQimax​ αντί για σταθερή τάση, ενώ η τάση του ζυγού-πιλότου είναι δεδομένη και το επίπεδο qqq είναι πρόσθετος άγνωστος. Οι παραλλαγές αυτές επιλύονται με τη μέθοδο Newton–Raphson και αριθμητικό υπολογισμό του ιακωβιανού πίνακα. Το πρόσθετο φορτίο συνδέεται τη χρονική στιγμή 20 s.

# --- παράμετροι της δευτερεύουσας ρύθμισης ---
V3_REF = 0.99            # τάση αναφοράς του ζυγού-πιλότου (p.u.)
K_P, K_I = 10.0, 0.3     # αναλογικός και ολοκληρωτικός όρος της ζώνης
K_L = 0.1                # τοπικός βρόχος: p.u. τάσης ανά p.u. Q και s
DT = 1.0                 # βήμα της προσομοίωσης (s)


def newton(F, x0, tol=1e-10, h=1e-7):
    """Newton–Raphson με αριθμητικό ιακωβιανό πίνακα (εμπρός διαφορές)."""
    x = np.array(x0, float)
    for _ in range(30):
        f = F(x)
        if np.abs(f).max() < tol:
            return x
        J = np.column_stack([(F(x + h * e) - f) / h
                             for e in np.eye(len(x))])
        x = x - np.linalg.solve(J, f)
    raise RuntimeError("η επίλυση δεν συγκλίνει")


def monimi_svr(lam=1.0, n_pyk=N_PYK, V3_ref=V3_REF, dP3=0.0, dQ3=0.0):
    """Μόνιμη κατάσταση με δευτερεύουσα ρύθμιση: V3 = V3_ref και
    Q_i = q Q_i^max. Άγνωστοι: θ2..θ5, V2..V5 και το επίπεδο q."""
    PL, QL = lam * P_L, lam * Q_L
    PL[2] += dP3
    QL[2] += dQ3
    P_sp = -PL
    for g, d in MONADES.items():
        P_sp[g] += d["P"]
    Y = build_ybus(KLADOI, n_pyk)

    def F(x):
        th, V, q = np.r_[0.0, x[:4]], np.r_[V1, x[4:8]], x[8]
        S = egxyseis(Y, V, th)
        Q_G = S.imag + QL
        return np.r_[S.real[1:] - P_sp[1:],     # ισοζύγια ενεργής ισχύος
                     Q_G[1:3],                 # ζυγοί 2, 3 χωρίς παραγωγή
                     Q_G[GEN] - q * QMAX,      # κοινό επίπεδο q
                     V[2] - V3_ref]            # τάση του ζυγού-πιλότου

    x = newton(F, np.r_[np.zeros(4), np.ones(4), 0.5])
    th, V, q = np.r_[0.0, x[:4]], np.r_[V1, x[4:8]], x[8]
    return V, th, egxyseis(Y, V, th), q


def prosomoiosi(defterevousa=True, T=300.0, t_bima=20.0):
    """Οιονεί μόνιμη προσομοίωση της πρωτεύουσας και της δευτερεύουσας.

    Στήλες του αποτελέσματος: t, V3, V4, V5, Q4/Q4max, Q5/Q5max.
    """
    V0, _, _, q = monimi_svr()
    V_ref = V0.copy()                    # τιμές αναφοράς των AVR
    olokl = q / K_I                      # ολοκλήρωμα του σφάλματος
    apot = []
    for k in range(int(T / DT) + 1):
        t = k * DT
        bima = float(t >= t_bima)
        V, th, S, Q_G = katastasi(V_ref=V_ref, dP3=bima * dP3,
                                  dQ3=bima * dQ3)
        apot.append([t, V[2], *V[GEN], *(Q_G[GEN] / QMAX)])
        if defterevousa:
            e = V3_REF - V[2]
            q_nea = K_P * e + K_I * (olokl + e * DT)
            if -0.5 < q_nea < 1.0:       # χωρίς ολοκλήρωση στον κορεσμό
                olokl += e * DT
            q = float(np.clip(q_nea, -0.5, 1.0))
            V_ref[GEN] += DT * K_L * (q * QMAX - Q_G[GEN])
    return np.array(apot)

Στο τέλος οι δύο μονάδες διαθέτουν το ίδιο ποσοστό εφεδρείας, ώστε καμία να μη φθάσει πρώτη στο όριό της σε ένα επόμενο συμβάν.

Το σχήμα δείχνει την τελική κατάσταση με τη δευτερεύουσα ρύθμιση και, σε παρένθεση, τις μεταβολές ως προς την τελική κατάσταση με μόνη την πρωτεύουσα ρύθμιση. Για να αυξηθεί η τάση του ζυγού-πιλότου κατά 0,01 p.u., η άεργη ισχύς της G4 αυξάνεται κατά 11 Mvar περίπου, ενώ της G5 μειώνεται ελάχιστα. Επειδή η τάση του ζυγού 1 παραμένει σταθερή, σχεδόν όλη η πρόσθετη άεργη ισχύς ρέει προς το υπόλοιπο σύστημα, και η τάση στους ακροδέκτες της G4 υπερβαίνει το όριο του 1,05 p.u. Τέτοιες καταστάσεις αποφεύγει η τριτεύουσα ρύθμιση, στην οποία οι περιορισμοί τάσης περιλαμβάνουν και τους ακροδέκτες των μονάδων.

Τριτεύουσα ρύθμιση: προγραμματισμός της επόμενης ημέρας¶

Κατά τον προγραμματισμό της επόμενης ημέρας, ο ΔΣΜΗΕ καθορίζει για κάθε αγοραία χρονική μονάδα τις τιμές αναφοράς της δευτερεύουσας ρύθμισης και το πρόγραμμα χειρισμών των πυκνωτών, με βάση την πρόβλεψη του φορτίου. Οι μεταβλητές ελέγχου του προβλήματος (11) είναι εδώ η τάση αναφοράς του ζυγού-πιλότου και τα βήματα πυκνωτών σε λειτουργία, και η κατάσταση για κάθε συνδυασμό τους είναι η μόνιμη κατάσταση της δευτερεύουσας ρύθμισης. Οι περιορισμοί είναι 0,95≤Vi≤1,050{,}95 \le V_i \le 1{,}050,95≤Vi​≤1,05 p.u. σε όλους τους ζυγούς, μαζί με τους ακροδέκτες των μονάδων, και −0,5≤q≤1-0{,}5 \le q \le 1−0,5≤q≤1, ώστε η άεργη ισχύς των μονάδων να παραμένει εντός των ορίων τους. Επειδή τα βήματα των πυκνωτών είναι διακριτά, το πρόβλημα επιλύεται με εξαντλητική αναζήτηση (exhaustive search): δοκιμάζονται όλοι οι συνδυασμοί της τάσης αναφοράς, ανά 0,01 p.u., και του αριθμού των βημάτων σε λειτουργία, και επιλέγεται ο καλύτερος εφικτός. Εξετάζονται δύο κριτήρια: οι απώλειες ενεργής ισχύος της ζώνης και οι απώλειες με έναν πρόσθετο όρο w q2w\,q^2wq2, ο οποίος ευνοεί τη διατήρηση εφεδρείας άεργης ισχύος στις μονάδες.

Τριτεύουσα ρύθμιση ως πρόβλημα MINLP¶

Η εξαντλητική αναζήτηση εξετάζει την τάση αναφοράς μόνο ανά 0,01 p.u., και το πλήθος των συνδυασμών αυξάνεται εκθετικά με τον αριθμό των διακριτών μέσων. Το ίδιο πρόβλημα διατυπώνεται ως πρόβλημα μεικτού ακέραιου μη γραμμικού προγραμματισμού (mixed-integer nonlinear programming, MINLP), στο οποίο συνυπάρχουν συνεχείς και ακέραιες μεταβλητές. Συνεχείς μεταβλητές είναι οι γωνίες θi\theta_iθi​ και τα μέτρα ViV_iVi​ των τάσεων των ζυγών 2–5 και το επίπεδο qqq, ενώ ακέραια είναι το πλήθος nnn των βημάτων πυκνωτών σε λειτουργία:

from math import ceil, floor
from scipy.optimize import minimize


def xalarosi(lam, w, n_kato, n_pano, x0):
    """Χαλάρωση του MINLP, με το n συνεχές στο [n_kato, n_pano].

    Μεταβλητές x = [θ2, θ3, θ4, θ5, V2, V3, V4, V5, q, n].
    """
    PL, QL = lam * P_L, lam * Q_L
    P_sp = -PL
    for g, d in MONADES.items():
        P_sp[g] += d["P"]
    Y0 = build_ybus(KLADOI)                   # χωρίς πυκνωτές

    def egxyseis_x(x):
        th, V = np.r_[0.0, x[:4]], np.r_[V1, x[4:8]]
        Y = Y0.copy()
        Y[2, 2] += 1j * x[9] * B_VIMA         # n βήματα πυκνωτών
        return egxyseis(Y, V, th)

    def kritirio(x):                          # απώλειες (p.u.) + w q²
        return egxyseis_x(x).real.sum() + w * x[8] ** 2

    def isotites(x):
        S = egxyseis_x(x)
        Q_G = S.imag + QL
        return np.r_[S.real[1:] - P_sp[1:],   # ισοζύγια ενεργής ισχύος
                     Q_G[1:3],                # ζυγοί 2, 3 χωρίς παραγωγή
                     Q_G[GEN] - x[8] * QMAX]  # κοινό επίπεδο q

    oria = ([(None, None)] * 4 + [(V_MIN, V_MAX)] * 4   # γωνίες, τάσεις
            + [(-0.5, 1.0), (n_kato, n_pano)])          # q, n
    return minimize(kritirio, x0, method="SLSQP", bounds=oria,
                    constraints={"type": "eq", "fun": isotites},
                    options={"ftol": 1e-12, "maxiter": 300})


def diaklados(lam, w):
    """Μέθοδος διακλάδωσης και φραγής ως προς το πλήθος βημάτων n."""
    x0 = np.r_[np.zeros(4), np.ones(4), 0.0, 2.0]       # αρχική εκτίμηση
    J_kal, x_kal = np.inf, None               # καλύτερη ακέραια λύση
    anoixta = [(0, max(BIMATA))]              # κλάδοι προς επίλυση
    while anoixta:
        n_kato, n_pano = anoixta.pop()
        res = xalarosi(lam, w, n_kato, n_pano, x0)
        n = res.x[9]
        if not res.success or res.fun >= J_kal:
            continue                          # αδύνατη ή χωρίς βελτίωση
        if abs(n - round(n)) < 1e-6:
            J_kal, x_kal = res.fun, res.x     # νέα καλύτερη ακέραια λύση
        else:                                 # διακλάδωση
            anoixta += [(n_kato, floor(n)), (ceil(n), n_pano)]
    return x_kal, J_kal

Καμπύλες P–V και περιθώριο φόρτισης¶

Τα φορτία των ζυγών 2 και 3 αυξάνονται αναλογικά, με συντελεστή φόρτισης λ\lambdaλ και σταθερό συντελεστή ισχύος, και η πρόσθετη ενεργή ισχύς εισάγεται από το υπόλοιπο σύστημα. Οι μονάδες διατηρούν την τάση των ακροδεκτών τους μέχρι η άεργη ισχύς τους να φθάσει στο Qmax⁡Q^{\max}Qmax. Από εκεί και πέρα η άεργη ισχύς τους παραμένει σταθερή και η τάση τους μειώνεται, δηλαδή ο ζυγός της μονάδας συμπεριφέρεται ως ζυγός φορτίου, όπως όταν ο περιοριστής υπερδιέγερσης περιορίζει τη διέγερση.

Η καμπύλη PPP–VVV υπολογίζεται με μια απλή μορφή επαναληπτικών ροών ισχύος. Η τάση V3V_3V3​ του ζυγού 3 μειώνεται σε μικρά βήματα και για κάθε τιμή της επιλύονται τα ισοζύγια ισχύος με άγνωστο, μαζί με τις γωνίες και τις υπόλοιπες τάσεις, τον συντελεστή φόρτισης λ\lambdaλ. Σε αντίθεση με τη συνήθη ροή ισχύος, ο ιακωβιανός πίνακας της διατύπωσης αυτής δεν γίνεται ιδιάζων στη μύτη, οπότε η καμπύλη υπολογίζεται και πέρα από αυτήν. Η συνάρτηση isozygia επιλύει τα ισοζύγια με δεδομένη V3V_3V3​ και η sarosi εκτελεί τη σάρωση με έλεγχο των ορίων άεργης ισχύος. Η ίδια διατύπωση χρησιμοποιείται στη συνέχεια και για τις καμπύλες QQQ–VVV.

def isozygia(Y, V3, V, th, u, eleft, orio, P_G, pv=True, lam=1.0):
    """Ισοζύγια ισχύος με δεδομένη τάση V3 στον ζυγό 3.

    Άγνωστοι: θ2..θ5, οι τάσεις των ζυγών eleft και το u, το οποίο
    είναι ο συντελεστής φόρτισης λ (pv=True, καμπύλη P–V) ή η άεργη
    ισχύς Qc υποθετικού συστήματος αντιστάθμισης στον ζυγό 3 (pv=False,
    καμπύλη Q–V με φόρτιση lam). orio: μονάδες στο όριο άεργης ισχύος.
    """
    m = len(eleft)

    def anaptyxi(x):
        V_, th_ = V.copy(), np.r_[0.0, x[:4]]
        V_[2], V_[eleft] = V3, x[4:4 + m]
        lam_, Qc = (x[-1], 0.0) if pv else (lam, x[-1])
        return V_, th_, lam_, Qc

    def F(x):
        V_, th_, lam_, Qc = anaptyxi(x)
        S = egxyseis(Y, V_, th_)
        dP = S.real[1:] - (P_G[1:] - lam_ * P_L[1:])
        dQ = [S.imag[i] - MONADES[i]["Qmax"] if i in orio
              else S.imag[i] + lam_ * Q_L[i] - (Qc if i == 2 else 0.0)
              for i in eleft + [2]]
        return np.r_[dP, dQ]

    return anaptyxi(newton(F, np.r_[th[1:], V[eleft], u]))


def sarosi(V3s, ektos=(), xoris=(), n_pyk=N_PYK, pv=True, lam=1.0,
           lam0=0.6):
    """Διαδοχικές λύσεις για φθίνουσες τιμές της V3, με έλεγχο των
    ορίων άεργης ισχύος. Επιστρέφει λίστα (V, θ, λ, Qc, μονάδες στο όριο).
    """
    Y = build_ybus(KLADOI, n_pyk, ektos)
    entos = [g for g in GEN if g not in xoris]
    P_G = np.zeros(NB)
    for g in entos:
        P_G[g] = MONADES[g]["P"]
    V, th, _, _ = katastasi(lam0 if pv else lam, n_pyk, ektos=ektos,
                            xoris=xoris)
    u, orio, apot = (lam0 if pv else 0.0), [], []
    for V3 in V3s:
        while True:
            eleft = [1] + sorted(list(xoris) + orio)
            try:
                V, th, lam_, Qc = isozygia(Y, V3, V, th, u, eleft, orio,
                                           P_G, pv, lam)
            except (RuntimeError, np.linalg.LinAlgError):
                return apot
            Q_G = egxyseis(Y, V, th).imag
            nea = [g for g in entos
                   if g not in orio and Q_G[g] > MONADES[g]["Qmax"]]
            if not nea:
                break
            orio += nea                  # η μονάδα παραμένει στο όριο
        u = lam_ if pv else Qc
        apot.append((V.copy(), th.copy(), lam_, Qc, tuple(orio)))
        if pv and lam_ < lam0:          # τέλος του κάτω τμήματος
            break
    return apot

Ανάλυση ιδιομορφών του μειωμένου ιακωβιανού πίνακα¶

Ο μειωμένος ιακωβιανός πίνακας της (17) υπολογίζεται από τις παραγώγους της ροής ισχύος, με ζυγούς φορτίου τους ζυγούς 2 και 3 και τους ζυγούς των μονάδων που βρίσκονται στο όριό τους. Οι ιδιοτιμές του, τα δεξιά ιδιοδιανύσματα ξi\boldsymbol{\xi}_iξi​ και τα αριστερά ηi\boldsymbol{\eta}_iηi​ δίνουν τους συντελεστές συμμετοχής pki=ξki ηikp_{ki} = \xi_{ki}\,\eta_{ik}pki​=ξki​ηik​ του ζυγού kkk στην ιδιομορφή iii, ενώ τα διαγώνια στοιχεία του JR−1\mathbf{J}_R^{-1}JR−1​ είναι οι ευαισθησίες ∂Vk/∂Qk\partial V_k/\partial Q_k∂Vk​/∂Qk​.

def meiomenos(V, th, pq, n_pyk=N_PYK, ektos=()):
    """Μειωμένος ιακωβιανός πίνακας J_R = J_QV - J_Qθ J_Pθ^-1 J_PV."""
    Y = build_ybus(KLADOI, n_pyk, ektos)
    dSt, dSv = paragogoi(Y, V, th)
    an = list(range(1, NB))
    J_Pth = dSt.real[np.ix_(an, an)]
    J_PV = dSv.real[np.ix_(an, pq)]
    J_Qth = dSt.imag[np.ix_(pq, an)]
    J_QV = dSv.imag[np.ix_(pq, pq)]
    return J_QV - J_Qth @ np.linalg.solve(J_Pth, J_PV)


def idiomorfes(J_R):
    w, xi = np.linalg.eig(J_R)
    eta = np.linalg.inv(xi)                  # αριστερά ιδιοδιανύσματα
    seira = np.argsort(w.real)
    p = (xi * eta.T).real                    # p[k, i] = ξ_ki η_ik
    return w.real[seira], p[:, seira]

def isorropia(n, X, P0, alpha=ALPHA, V_prin=1.0, E=E):
    """Τάση V του δικτύου μεταφοράς στο σημείο ισορροπίας, για πηγή
    τάσης E πίσω από αντίδραση X.

    Αναζητείται η υψηλότερη τομή κάτω από την προηγούμενη τάση, όπου η
    ισχύς του δικτύου ξεπερνά τη ζήτηση. Επιστρέφει nan αν δεν υπάρχει.
    """
    t = TAN_PHI
    k = E**2 * (1 + t * t)

    def g(V):
        p_diktyo = V * (-t * V + np.sqrt(np.maximum(k - V**2, 0.0)))
        return p_diktyo / (X * (1 + t * t)) - P0 * (n * V)**alpha

    Vs = np.arange(min(V_prin + 0.02, np.sqrt(k)), 0.0, -0.0005)
    gs = g(Vs)
    idx = np.flatnonzero((gs[:-1] < 0) & (gs[1:] >= 0))
    if idx.size == 0:
        return np.nan
    a, b = Vs[idx[0]], Vs[idx[0] + 1]
    for _ in range(40):                      # διχοτόμηση
        m = (a + b) / 2
        a, b = (m, b) if g(m) < 0 else (a, m)
    return (a + b) / 2


def hed(V, n, X, P0, alpha=ALPHA):
    """ΗΕΔ E_q της γεννήτριας, όταν ο AVR κρατά την τάση των ακροδεκτών
    της ίση με E."""
    P = P0 * (n * V)**alpha
    delta = np.arcsin(P * X / (E * V))       # γωνία της E ως προς τη V
    I = (E - V * np.exp(-1j * delta)) / (1j * X)
    return abs(E + 1j * X_D * I)


def prosomoiosi_oltc(P0=0.85, alpha=ALPHA, V_mplok=0.0, T=300.0,
                     t_sym=10.0, dt=1.0):
    """Οιονεί μόνιμη προσομοίωση: απώλεια γραμμής τη στιγμή t_sym,
    χειρισμοί του ΣΑΤΥΦ και περιοριστής υπερδιέγερσης. V_mplok: τάση
    κάτω από την οποία αναστέλλονται οι χειρισμοί (0: χωρίς αναστολή).
    Στήλες του αποτελέσματος: t, V, V_L, n, περιοριστής σε λειτουργία."""
    n, xron, protos, V = 1.0, 0.0, True, 1.0
    oel, xron_oel = False, 0.0
    apot = []
    for k in range(int(T / dt) + 1):
        t = k * dt
        X = X_PRIN if t < t_sym else X_META
        if oel:                              # E_q = EQ_MAX πίσω από X_D + X
            V = isorropia(n, X_D + X, P0, alpha, V, E=EQ_MAX)
        else:                                # ο AVR κρατά την τάση E
            V = isorropia(n, X, P0, alpha, V)
        if np.isnan(V):                      # δεν υπάρχει ισορροπία
            break
        VL = n * V
        apot.append((t, V, VL, n, oel))
        if not oel:                          # περιοριστής υπερδιέγερσης
            yperbasi = hed(V, n, X, P0, alpha) > EQ_MAX
            xron_oel = xron_oel + dt if yperbasi else 0.0
            oel = xron_oel >= T_OEL
        if abs(VL - 1.0) > NEKRI and V >= V_mplok:
            xron += dt
            if xron >= (T1 if protos else T2):
                bima = N_BIMA if VL < 1.0 else -N_BIMA
                n = float(np.clip(n + bima, N_MIN, N_MAX))
                xron, protos = 0.0, False
        else:
            xron, protos = 0.0, True
    return np.array(apot)

import numpy as np
import matplotlib.pyplot as plt

# --- παράμετροι: σημεία σύνδεσης, βύθιση και μέγιστο ρεύμα μετατροπέα ---
SCR = {"ισχυρό": 5.0, "ασθενές": 2.5, "πολύ ασθενές": 1.5}
U_f = 0.5
I_MAX = 1.1


def stirixi(SCR, k, U_f=U_f, I_max=I_MAX):
    """Τάση, άεργο και ενεργό ρεύμα και ενεργή ισχύς κατά τη βύθιση."""
    U = (U_f + k / SCR) / (1 + k / SCR)
    if k * (1 - U) > I_max:                  # όριο ρεύματος
        U = U_f + I_max / SCR
    I_q = min(k * (1 - U), I_max)
    I_p = min(1.0, np.sqrt(I_max**2 - I_q**2))
    return U, I_q, I_p, U * I_p
